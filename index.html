// Edge Function name: game-action // Answers yes/no questions and checks guesses against the target's secret character. import { createClient } from "npm:@supabase/supabase-js@2"; const admin = createClient( Deno.env.get("SUPABASE_URL")!, JSON.parse(Deno.env.get("SUPABASE_SECRET_KEYS")!)["default"], ); const cors = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", "Access-Control-Allow-Methods": "POST, OPTIONS", }; function json(body: unknown, status = 200) { return new Response(JSON.stringify(body), { status, headers: { ...cors, "Content-Type": "application/json" }, }); } Deno.serve(async (req) => { if (req.method === "OPTIONS") return new Response("ok", { headers: cors }); if (req.method !== "POST") return json({ error: "method not allowed" }, 405); // 1) who is calling const token = (req.headers.get("Authorization") ?? "").replace("Bearer ", ""); const { data: u, error: uErr } = await admin.auth.getUser(token); if (uErr || !u.user) return json({ error: "unauthorized" }, 401); const me = u.user.id; // 2) input let body: any; try { body = await req.json(); } catch { return json({ error: "bad json" }, 400); } const { room_id, action, target_user_id } = body; if (!room_id || !target_user_id || !["ask", "guess"].includes(action)) { return json({ error: "bad request" }, 400); } if (target_user_id === me) return json({ error: "cannot target yourself" }, 400); // 3) room checks const { data: room } = await admin.from("rooms").select("*").eq("id", room_id).single(); if (!room) return json({ error: "room not found" }, 404); if (room.status !== "playing") return json({ error: "game not running" }, 409); if (room.current_turn !== me) return json({ error: "not your turn" }, 403); const { data: players } = await admin .from("room_players") .select("user_id, joined_at") .eq("room_id", room_id) .order("joined_at"); const ids = (players ?? []).map((p) => p.user_id); if (!ids.includes(me) || !ids.includes(target_user_id)) { return json({ error: "not in room" }, 403); } // 4) the target's secret character (never sent to the client) const { data: secret } = await admin .from("secrets") .select("character_id") .eq("room_id", room_id) .eq("user_id", target_user_id) .single(); if (!secret) return json({ error: "target has no secret character" }, 409); const { data: ch } = await admin .from("characters") .select("id, name, series, attrs") .eq("id", secret.character_id) .single(); if (!ch) return json({ error: "character missing" }, 500); const next = ids[(ids.indexOf(me) + 1) % ids.length]; // 5a) a question if (action === "ask") { const { attr, value, label } = body; if ( typeof attr !== "string" || !/^[a-z_]{1,30}$/.test(attr) || typeof value !== "string" || value.length > 50 || typeof label !== "string" || label.length > 120 ) { return json({ error: "bad question" }, 400); } const { count } = await admin .from("moves") .select("id", { count: "exact", head: true }) .eq("room_id", room_id) .eq("user_id", me) .eq("kind", "question"); if ((count ?? 0) >= room.max_questions) { return json({ error: "no questions left" }, 409); } const yes = attr === "series" ? ch.series === value : ch.attrs?.[attr] === value; const answer = yes ? "نعم" : "لا"; await admin.from("moves").insert([ { room_id, user_id: me, kind: "question", content: label }, { room_id, user_id: target_user_id, kind: "answer", content: answer }, ]); await admin.from("rooms").update({ current_turn: next }).eq("id", room_id); return json({ answer }); } // 5b) a final guess const { guess_character_id } = body; if (typeof guess_character_id !== "string") return json({ error: "bad guess" }, 400); const { data: g } = await admin .from("characters") .select("id, name") .eq("id", guess_character_id) .single(); const correct = guess_character_id === ch.id; await admin.from("moves").insert([ { room_id, user_id: me, kind: "guess", content: g?.name ?? "؟" }, { room_id, user_id: target_user_id, kind: "answer", content: correct ? "صحيح" : "خطأ" }, ]); if (correct) { await admin.from("rooms").update({ status: "finished", current_turn: null }).eq("id", room_id); return json({ correct: true, character: ch.name }); } await admin.from("rooms").update({ current_turn: next }).eq("id", room_id); return json({ correct: false }); });
